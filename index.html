/* Page logic below — normally you only edit the configuration above. */
const cards = document.getElementById("cards");
const search = document.getElementById("search");
const platformFilter = document.getElementById("platformFilter");
const testFilter = document.getElementById("testFilter");
const resultCount = document.getElementById("resultCount");
const empty = document.getElementById("empty");
const grandFilters = document.getElementById("grandFilters");
let activeGrand = "all";
function escapeHtml(value) {
return String(value ?? "").replace(/[&<>"']/g, char => ({"&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;"}[char]));
}
function list(value) { return !value ? [] : (Array.isArray(value) ? value : [value]).map(String); }
function fillSelect(select, options, label) {
select.innerHTML = `<option value="all">${escapeHtml(label)}</option>` + options.map(v => `<option value="${escapeHtml(v)}">${escapeHtml(v)}</option>`).join("");
}
fillSelect(platformFilter, PLATFORM_OPTIONS, "All platforms");
fillSelect(testFilter, TEST_TYPE_OPTIONS, "All test types");
function matchesFilters(item) {
const query = search.value.trim().toLowerCase();
const types = list(item.testTypes), grandTests = list(item.grandTests);
const platformMatch = platformFilter.value === "all" || item.platform === platformFilter.value;
const typeMatch = testFilter.value === "all" || types.includes(testFilter.value);
const grandMatch = activeGrand === "all" || grandTests.includes(activeGrand);
const haystack = [item.title, item.platform, item.description, item.filename, ...list(item.exams), ...types, ...grandTests].join(" ").toLowerCase();
return platformMatch && typeMatch && grandMatch && haystack.includes(query);
}
function render() {
const visible = questionBanks.filter(matchesFilters);
cards.innerHTML = visible.map(item => {
const types = list(item.testTypes), grandTests = list(item.grandTests);
const tags = [...types.map(t => `<span class="tag">${escapeHtml(t)}</span>`), ...grandTests.map(t => `<span class="tag grand">${escapeHtml(t)}</span>`)];
if (!tags.length) tags.push('<span class="tag unset">Test type not set</span>');
const examText = list(item.exams).join(" · ") || "MCQ Bank";
return `<a class="card" href="${encodeURI(item.filename).replace(/'/g,"%27")}" style="--glow:${item.color || "#8b9cff"}">
<div class="card-top"><div class="platform-icon">${escapeHtml(item.icon || String(item.platform || "?").slice(0,1))}</div><div class="platform-name">${escapeHtml(item.platform)}</div></div>
<h3>${escapeHtml(item.title)}</h3><p>${escapeHtml(item.description || "Open this question bank and start practicing.")}</p>
<div class="tags">${tags.join("")}</div>
<div class="card-bottom"><span class="file">${escapeHtml(examText)} · ${escapeHtml(item.filename)}</span><span class="open">Open bank <span>↗</span></span></div></a>`;
}).join("");
resultCount.textContent = `${visible.length} ${visible.length === 1 ? "resource" : "resources"}`;
empty.style.display = visible.length ? "none" : "block";
document.getElementById("statCount").innerHTML = `${String(questionBanks.length).padStart(2,"0")} <span style="font-size:17px;color:#9aa8c2;font-weight:600">${questionBanks.length === 1 ? "file" : "files"}</span>`;
grandFilters.hidden = testFilter.value !== "Grand tests";
}
search.addEventListener("input", render);
platformFilter.addEventListener("change", render);
testFilter.addEventListener("change", () => {
activeGrand = "all";
grandFilters.querySelectorAll("[data-grand]").forEach(b => b.classList.toggle("active", b.dataset.grand === "all"));
render();
});
grandFilters.addEventListener("click", event => {
const button = event.target.closest("[data-grand]"); if (!button) return;
activeGrand = button.dataset.grand;
grandFilters.querySelectorAll("[data-grand]").forEach(b => b.classList.toggle("active", b === button));
render();
});
document.getElementById("randomButton").addEventListener("click", () => {
const eligible = questionBanks.filter(matchesFilters);
if (!eligible.length) { empty.style.display = "block"; empty.querySelector("strong").textContent = "No files match your current filters"; return; }
const picked = eligible[Math.floor(Math.random() * eligible.length)];
window.location.href = encodeURI(picked.filename).replace(/'/g, "%27");
});
render();
